% Accessor macros
\newcommand{\speccode}[1]{\csuse{spec@code@#1}}
\newcommand{\specdesc}[1]{\csuse{spec@desc@#1}}

% Records unique pages and their anchors
\newcommand{\specaddoccurrence}[3]{% #1 = key, #2 = page, #3 = label
	\ifinlistcs{#2}{spec@pages@#1}{}{%
		\listcsgadd{spec@pages@#1}{#2}%
		\listcsgadd{spec@labels@#1}{#3}%
	}%
}

% Formats and displays the list of pages (e.g., "Page 2" or "Pages 2, 4")
\newcounter{spec@pagecount}
\newcommand{\specpages}[1]{%
	\ifcsname spec@labels@#1\endcsname%
		\setcounter{spec@pagecount}{0}%
		\renewcommand*{\do}[1]{\stepcounter{spec@pagecount}}%
		\dolistcsloop{spec@labels@#1}%
		\ifnum\value{spec@pagecount}=0
			Page ??%
		\else
			\ifnum\value{spec@pagecount}>1 Pages\else Page\fi~%
			\def\spec@delim{}%
			\renewcommand*{\do}[1]{\spec@delim\pageref{##1}\def\spec@delim{, }}%
			\dolistcsloop{spec@labels@#1}%
		\fi
	\else%
		Page ??%
	\fi%
}

% Places an anchor and writes the occurrence
\newcommand{\speclabel}[1]{%
	\ifcsname spec@code@#1\endcsname%
		\stepcounter{spec@label@count}%
		\edef\spec@thislabel{spec:#1@\the\value{spec@label@count}}%
		\noindent{\footnotesize\ttfamily\color{black!55}%
			\phantomsection\label{\spec@thislabel}%
			\protected@write\@auxout{}{\string\specaddoccurrence{#1}{\thepage}{\spec@thislabel}}%
			[\csuse{spec@code@#1}]\ \textit{\csuse{spec@desc@#1}}%
		}%
	\else%
		\PackageWarning{spec_engine}{Unknown specification key '#1'}%
	\fi%
}

% Generates a table row
\newcommand{\specrow}[1]{%
	\texttt{\speccode{#1}} & \specdesc{#1} & \specpages{#1} \\
}
