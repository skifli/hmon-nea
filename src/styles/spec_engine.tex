\newcommand{\pii}[2][REDACTED]{%

		\settowidth{\piiwidth}{\footnotesize\ttfamily #1}%

	\fi%
}

		\raisebox{0.15ex}{\color{black}\rule{\piiwidth}{1.1ex}}%
	\else%
		\href{#2}{\texttt{#2}}%
	\fi%
}
